#include <cub/detail/warpspeed/squad/squad.cuh>
#include <cuda_runtime.h>
#include <cstdio>
#include <cstdlib>
#include <initializer_list>

void check(cudaError_t error)
{
  if (error != cudaSuccess)
  {
    std::fprintf(stderr, "%s\n", cudaGetErrorString(error));
    std::exit(1);
  }
}

__global__ void record_ranks(int* output)
{
  constexpr cub::detail::warpspeed::SquadDesc squads[]{{0, 1}, {1, 3}, {2, 2}, {3, 1}};
  const int linear_thread = static_cast<int>((threadIdx.z * blockDim.y + threadIdx.y) * blockDim.x + threadIdx.x);
  const auto registers = cub::detail::warpspeed::getSpecialRegisters();
  cub::detail::warpspeed::squadDispatch(registers, squads, [&](auto squad) {
    output[3 * linear_thread] = squad.mSquadIdx;
    output[3 * linear_thread + 1] = squad.warpRank();
    output[3 * linear_thread + 2] = squad.threadRank();
  });
}

int main()
{
  constexpr int threads = 7 * 32;
  int* device{};
  check(cudaMalloc(&device, 3 * threads * sizeof(int)));
  for (const bool two_dimensions : {false, true})
  {
    const dim3 shape = (two_dimensions) ? dim3{32, 7} : dim3{threads};
    record_ranks<<<1, shape>>>(device);
    check(cudaGetLastError());
    int host[3 * threads]{};
    check(cudaMemcpy(host, device, sizeof(host), cudaMemcpyDeviceToHost));
    std::printf("shape=%s\n", (two_dimensions) ? "(32,7,1)" : "(224,1,1)");
    for (int warp = 0; warp < 7; ++warp)
    {
      const int i = 3 * warp * 32;
      std::printf("physical warp %d: descriptor=%d warpRank=%d firstThreadRank=%d\n", warp, host[i], host[i+1], host[i+2]);
    }
  }
  check(cudaFree(device));
}
